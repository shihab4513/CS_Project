sudo apt update
sudo apt install -y strace python3-venv nodejs npm
mkdir -p ~/package-lab
cd ~/package-lab
cd ~/package-lab
source venv/bin/activate
python -m pip install --no-cache-dir "requests==2.32.3"
python -c "import requests; print(requests.__version__)"
It should print:
2.32.3
Then create a valid import trace:
strace -f -qq -e trace=process,file,network -s 256 -o requests_import.strace python -c "import requests; print(requests.__version__)"
Optional but recommended, create a valid safe-network trace:
strace -f -qq -e trace=process,file,network -s 256 -o requests_network.strace python -c "import requests; r=requests.get('https://example.com', timeout=10); print(r.status_code)"
It should normally print 200.
Now create a new archive containing only successful traces:
cd ~/package-lab
tar -czf safe_trace_logs_complete.tar.gz requests_import.strace requests_network.strace lodash-test/lodash_runtime.strace
ls -lh safe_trace_logs_complete.tar.gzcd ~/package-lab
